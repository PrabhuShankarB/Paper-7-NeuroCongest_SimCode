namespace ns3 {

Ns3Interface::Ns3Interface()
{
    m_rttWireless = 0.0;
    m_rttQueue = 0.0;
    m_confidence = 0.0;
    m_volatility = 0.0;

    m_selectedArm = 0;
    m_selectedCwnd = 65536;

    m_targetBsId = "";
}

Ns3Interface::~Ns3Interface()
{
}

std::vector<double>
Ns3Interface::BuildStateVector(
    double cwnd,
    double rtt,
    double lossRate,
    double throughput,
    double rsrp,
    double snr,
    double doppler,
    double queueDepth)
{
    std::vector<double> state;

    state.push_back(cwnd);
    state.push_back(rtt);
    state.push_back(lossRate);
    state.push_back(throughput);
    state.push_back(rsrp);
    state.push_back(snr);
    state.push_back(doppler);
    state.push_back(queueDepth);

    return state;
}

bool
Ns3Interface::DetectHandover(
    double currentRsrp,
    double threshold)
{
    if(currentRsrp < threshold)
    {
        return true;
    }

    return false;
}

void
Ns3Interface::RunCausalInference(
    const std::vector<double>& signalVector)
{
    /*
     * DRTD-CD
     *
     * signalVector:
     * RTT
     * RSRP
     * SNR
     * Doppler
     * ACK_gap
     */

    double rtt = signalVector[0];

    m_rttWireless = 0.35 * rtt;

    m_rttQueue =
        std::max(
            0.0,
            rtt - m_rttWireless);

    m_confidence = 0.91;
}
