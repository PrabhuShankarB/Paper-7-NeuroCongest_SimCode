#ifndef TCP_NEUROCONGEST_H
#define TCP_NEUROCONGEST_H

#include "ns3/tcp-congestion-ops.h"
#include "ns3/traced-value.h"
#include "ns3/event-id.h"

namespace ns3 {

class TcpNeuroCongest : public TcpCongestionOps
{
public:

    static TypeId GetTypeId (void);

    TcpNeuroCongest ();

    TcpNeuroCongest (const TcpNeuroCongest& sock);

    virtual ~TcpNeuroCongest ();

    virtual std::string GetName () const override;

    virtual void IncreaseWindow (
        Ptr<TcpSocketState> tcb,
        uint32_t segmentsAcked) override;

    virtual uint32_t GetSsThresh (
        Ptr<const TcpSocketState> state,
        uint32_t bytesInFlight) override;

    virtual void PktsAcked (
        Ptr<TcpSocketState> tcb,
        uint32_t segmentsAcked,
        const Time& rtt) override;

    virtual Ptr<TcpCongestionOps> Fork () override;

private:

    void CollectState(
        Ptr<TcpSocketState> tcb,
        const Time& rtt);

    void InvokeNeuroCongest();

    void ApplyCwndDecision(
        Ptr<TcpSocketState> tcb);

private:

    double m_rtt;

    double m_throughput;

    double m_lossRate;

    double m_rsrp;

    double m_snr;

    double m_doppler;

    double m_queueDepth;

    uint32_t m_selectedCwnd;

    EventId m_updateEvent;
};

}

#endif
