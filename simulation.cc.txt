#include "ns3/core-module.h"
#include "ns3/network-module.h"
#include "ns3/internet-module.h"
#include "ns3/mobility-module.h"
#include "ns3/flow-monitor-module.h"
#include "ns3/applications-module.h"

#include "topology.h"
#include "tcp-neurocongest.h"

using namespace ns3;

NS_LOG_COMPONENT_DEFINE("NeuroCongestSimulation");

int
main(int argc, char *argv[])
{
    /*
     * ==================================================
     * Simulation Parameters
     * ==================================================
     */

    uint32_t numMacro = 5;
    uint32_t numSmall = 20;
    uint32_t numFemto = 10;
    uint32_t numUe = 200;

    double simTime = 1000.0;

    /*
     * ==================================================
     * Create Topology
     * ==================================================
     */

    HetNetTopology topology;

    topology.CreateTopology();

    NodeContainer allBs;

    allBs.Add(topology.macroBs);
    allBs.Add(topology.smallBs);
    allBs.Add(topology.femtoBs);

    /*
     * ==================================================
     * Mobility Models
     * ==================================================
     */

    MobilityHelper bsMobility;

    bsMobility.SetMobilityModel(
        "ns3::ConstantPositionMobilityModel");

    bsMobility.Install(allBs);

    MobilityHelper ueMobility;

    ueMobility.SetMobilityModel(
        "ns3::RandomWaypointMobilityModel");

    ueMobility.Install(topology.ueNodes);

    /*
     * ==================================================
     * Internet Stack
     * ==================================================
     */

    InternetStackHelper internet;

    internet.Install(allBs);
    internet.Install(topology.ueNodes);

    /*
     * ==================================================
     * Install NeuroCongest TCP
     * ==================================================
     */

    Config::SetDefault(
        "ns3::TcpL4Protocol::SocketType",
        TypeIdValue(
            TcpNeuroCongest::GetTypeId()));

    /*
     * ==================================================
     * Traffic Generation
     * ==================================================
     */

    uint16_t port = 5000;

    BulkSendHelper source(
        "ns3::TcpSocketFactory",
        InetSocketAddress(
            Ipv4Address("10.0.0.1"),
            port));

    source.SetAttribute(
        "MaxBytes",
        UintegerValue(0));

    ApplicationContainer srcApps;

    for(uint32_t i = 0; i < numUe; i++)
    {
        srcApps.Add(
            source.Install(
                topology.ueNodes.Get(i)));
    }

    srcApps.Start(Seconds(1.0));
    srcApps.Stop(Seconds(simTime));

    /*
     * ==================================================
     * Packet Sink
     * ==================================================
     */

    PacketSinkHelper sink(
        "ns3::TcpSocketFactory",
        InetSocketAddress(
            Ipv4Address::GetAny(),
            port));

    ApplicationContainer sinkApps;

    sinkApps.Add(
        sink.Install(
            topology.macroBs.Get(0)));

    sinkApps.Start(Seconds(0.0));
    sinkApps.Stop(Seconds(simTime));

    /*
     * ==================================================
     * Flow Monitor
     * ==================================================
     */

    FlowMonitorHelper flowHelper;

    Ptr<FlowMonitor> monitor =
        flowHelper.InstallAll();

    /*
     * ==================================================
     * Run Simulation
     * ==================================================
     */

    Simulator::Stop(
        Seconds(simTime));

    Simulator::Run();

    /*
     * ==================================================
     * Export Metrics
     * ==================================================
     */

    monitor->SerializeToXmlFile(
        "neurocongest-results.xml",
        true,
        true);

    Simulator::Destroy();

    return 0;
}
